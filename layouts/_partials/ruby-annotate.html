{{/* Character string + Pinyin string → ruby fragment (no outer block tag) Parameters: dict "hanzi" string "pinyin" string Pinyin is split by whitespace into syllables; punctuation / whitespace / Latin numerals in the Chinese side do not consume syllables. */}} {{- $hanzi := .hanzi -}} {{- $pinyin := .pinyin -}} {{- $pyClean := replaceRE `[,.!?，。！？、；：·…—\-]` " " $pinyin -}} {{- $pyClean = replaceRE `\s+` " " (trim $pyClean " ") -}} {{- $syllables := split $pyClean " " -}} {{- $si := 0 -}} {{- $sylLen := len $syllables -}} {{- range split $hanzi "" -}} {{- $ch := . -}} {{- if ne $ch "" -}} {{- if findRE `^[\s，。！？、；：""''“”‘’（）()【】《》〈〉…—·.,!?;:'\"\[\]{}「」『』〔〕]$` $ch -}} {{ $ch }} {{- else if findRE `^[A-Za-z0-9]$` $ch -}} {{ $ch }} {{- else -}} {{- $rt := "" -}} {{- if lt $si $sylLen -}} {{- $rt = index $syllables $si -}} {{- $si = add $si 1 -}} {{- end -}} {{- if $rt -}} {{ $ch }}({{ $rt }}) {{- else -}} {{ $ch }} {{- end -}} {{- end -}} {{- end -}} {{- end -}}
