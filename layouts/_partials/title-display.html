between segments - If no : output as‑is - The caller is responsible for passing the final text to display (already retrieved from front matter) */ -}} {{- $text := . | emojify -}} {{- if in $text "" -}} {{- $parts := split $text "" -}} {{- $last := sub (len $parts) 1 -}} {{- range $idx, $part := $parts -}} {{- $part | safeHTML -}}{{- if lt $idx $last -}}
{{- end -}} {{- end -}} {{- else -}} {{- $text | safeHTML -}} {{- end -}}
